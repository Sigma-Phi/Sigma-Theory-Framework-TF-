import numpy as np

class NegentropicSystem:
    def __init__(self, size):
        self.state = np.random.rand(size) # 初始邏輯狀態
        self.entropy = 1.0
        
    def observe_state(self):
        """將理論變量映射為具體觀測指標"""
        logic_entropy = -np.sum(self.state * np.log(self.state + 1e-9))
        info_potential = 1.0 / (logic_entropy + 1e-9)
        return {"entropy": logic_entropy, "potential": info_potential}

    def update(self, external_input):
        # 模擬負熵引入對抗熵增
        noise = np.random.normal(0, 0.1, self.state.shape)
        # 核心動力學方程
        self.state += (external_input - self.state) * 0.1 + noise
        self.state = np.clip(self.state, 0, 1)

system = NegentropicSystem(10)
for _ in range(5):
    ext_signal = np.random.rand(10)
    system.update(ext_signal)

### 7️⃣ 討論 (DISCUSSION)
本模型與傳統資訊處理最大的區別在於「自我糾偏的非線性」。傳統系統傾向於通過增加計算算力來處理雜訊，而本模型引入了耗散結構思想，透過動態調整系統內部連通性來消耗雜訊的資訊能量，從而將物理層面的「熱力學秩序」轉化為邏輯層面的「計算秩序」。
### 8️⃣ 限制 (LIMITATIONS)
 * **邊界假設**：假設外部環境具備足夠的可提取負熵（Non-random Information）。
 * **數學難點**：在極度不穩定的資訊環境下，非線性反饋項可能導致邏輯震盪（Bifurcation），如何保證在動態分岔點的全局收斂性仍需進一步定義拓撲約束。
